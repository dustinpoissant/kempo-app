Form JavaScript
import DB from '/framework/src/renderer/utils/sqlDB.js';
import Toast from '/modules/kempo-ui/dist/components/Toast.js';
const contacts = new DB('contacts').table('contacts');
document.querySelector('#contacts-form').addEventListener('submit', async e => {
  e.preventDefault();
  const name = document.getElementById('contact-name').value.trim();
  if (!name) return;
  try {
    const val = id => document.getElementById(id).value.trim() || null;
    await contacts.create({
      name,
      email: val('contact-email'),
      phone: val('contact-phone'),
      address: val('contact-address'),
      gender: document.getElementById('contact-gender').value || null,
      birthday: document.getElementById('contact-birthday').value || null
    });
    Toast.success('Contact added');
    ['contact-name', 'contact-email', 'contact-phone', 'contact-address', 'contact-birthday'].forEach(id => document.getElementById(id).value = '');
    document.getElementById('contact-gender').value = '';
    window.dispatchEvent(new Event('contacts:reload'));
  } catch (err) {
    console.error(err);
    Toast.error(err.message.replace(/^Error invoking remote method '[^']+': /, ''));
  }
});

Table JavaScript
import DB from '/framework/src/renderer/utils/sqlDB.js';
import Toast from '/modules/kempo-ui/dist/components/Toast.js';
const contacts = new DB('contacts').table('contacts');
const toastErr = err => {
  console.error(err);
  Toast.error(err.message.replace(/^Error invoking remote method '[^']+': /, ''));
};
const fields = [
  { name: 'name', label: 'Name' },
  { name: 'email', label: 'Email' },
  { name: 'phone', label: 'Phone' },
  { name: 'address', label: 'Address' },
  { name: 'gender', label: 'Gender' },
  { name: 'birthday', label: 'Birthday', type: 'date' }
];
const loadContacts = async () => {
  const records = await contacts.getAll();
  document.getElementById('contacts-table').setData({ records: records.map(r => r.data), fields });
};
await loadContacts();
window.addEventListener('contacts:reload', loadContacts);

document.getElementById('contacts-table').addEventListener('requestSave', async e => {
  const { record, newData, approve, reject } = e.detail;
  try {
    await contacts.update({ id: record.id }, newData);
    approve();
    await loadContacts();
  } catch (err) {
    reject();
    toastErr(err);
  }
});

document.getElementById('contacts-table').addEventListener('requestDelete', async e => {
  const { records, approve, reject } = e.detail;
  try {
    for (const r of records) await contacts.deleteAll({ id: r.id });
    approve();
    await loadContacts();
  } catch (err) {
    reject();
    toastErr(err);
  }
});
